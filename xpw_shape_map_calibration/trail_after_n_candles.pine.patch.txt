// Trail after N same-direction candles: count the chart candles that close in
// the trade's direction, in a row, after the entry candle. Counted on closed
// candles only, so history and live agree.
var int  runCnt  = 0
var bool trailGo = false
var int  runFor  = na
if strategy.position_size == 0
    runCnt  := 0
    trailGo := false
    runFor  := na
else
    int entBar = strategy.opentrades.entry_bar_index(strategy.opentrades - 1)
    if na(runFor) or runFor != entBar
        runFor  := entBar
        runCnt  := 0
        trailGo := false
    if barstate.isconfirmed and bar_index > entBar
        bool sameDir = strategy.position_size > 0 ? close > open : close < open
        runCnt := sameDir ? runCnt + 1 : 0
        if runCnt >= trailAfterN
            trailGo := true
bool trailArmed = trailOn and tdTk > 0 and (trailAfterN == 0 or trailGo)
float xTa = trailArmed ? taTk : na
float xTd = trailArmed ? tdTk : na
